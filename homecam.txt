  <script>
    let peer = null;
    let localStream = null;
    let wakeLock = null;

    function switchMode(mode) {
      const isCamera = mode === 'camera';
      document.getElementById('tabCamera').classList.toggle('active', isCamera);
      document.getElementById('tabViewer').classList.toggle('active', !isCamera);
      document.getElementById('cameraSection').classList.toggle('hidden', !isCamera);
      document.getElementById('viewerSection').classList.toggle('hidden', isCamera);
    }

    // Keep the screen awake so the phone doesn't sleep
    async function requestWakeLock() {
      try {
        if ('wakeLock' in navigator) {
          wakeLock = await navigator.wakeLock.request('screen');
        }
      } catch (err) {
        console.warn("WakeLock error:", err);
      }
    }

    // --- HOME PHONE: BROADCASTER ---
    async function startBroadcasting() {
      const key = document.getElementById('secretKey').value.trim();
      const statusEl = document.getElementById('cameraStatus');
      const facing = document.getElementById('cameraFacing').value;

      if (!key) return alert("Please enter a Secret Key!");

      statusEl.textContent = "Starting camera...";
      try {
        // Request video with 720p resolution for smooth streaming on Wi-Fi
        localStream = await navigator.mediaDevices.getUserMedia({
          video: { facingMode: facing, width: { ideal: 1280 }, height: { ideal: 720 } },
          audio: false
        });

        const preview = document.getElementById('localPreview');
        preview.srcObject = localStream;
        preview.classList.remove('hidden');

        await requestWakeLock();

        // Connect to PeerJS with the secret key as our unique Peer ID
        peer = new Peer(key, {
          config: {
            iceServers: [
              { urls: 'stun:stun.l.google.com:19302' },
              { urls: 'stun:stun1.l.google.com:19302' }
            ]
          }
        });

        peer.on('open', (id) => {
          statusEl.textContent = `🟢 Broadcasting live! (ID: ${id})`;
        });

        // When office calls, answer with the video stream
        peer.on('call', (call) => {
          statusEl.textContent = "🔵 Viewer connected! Streaming live feed...";
          call.answer(localStream);
        });

        peer.on('error', (err) => {
          if (err.type === 'unavailable-id') {
            statusEl.textContent = "❌ That Key is already in use. Pick a more unique Secret Key.";
          } else {
            statusEl.textContent = `Error: ${err.message}`;
          }
        });

      } catch (err) {
        statusEl.textContent = `Camera Error: ${err.message}`;
      }
    }

    // --- OFFICE: VIEWER ---
    function connectToCamera() {
      const key = document.getElementById('secretKey').value.trim();
      const statusEl = document.getElementById('viewerStatus');
      if (!key) return alert("Please enter the Secret Key you used on your phone!");

      statusEl.textContent = "Connecting to Peer network...";

      // Viewer gets a random client ID
      peer = new Peer({
        config: {
          iceServers: [
            { urls: 'stun:stun.l.google.com:19302' },
            { urls: 'stun:stun1.l.google.com:19302' }
          ]
        }
      });

      peer.on('open', () => {
        statusEl.textContent = `Dialing home camera (${key})...`;
        // Make call to home phone
        const call = peer.call(key, new MediaStream());

        call.on('stream', (remoteStream) => {
          statusEl.textContent = "🟢 Connected! Live feed streaming.";
          const video = document.getElementById('remoteFeed');
          video.srcObject = remoteStream;
          video.play();
        });

        call.on('close', () => {
          statusEl.textContent = "Feed closed by camera.";
        });
      });

      peer.on('error', (err) => {
        statusEl.textContent = `Connection failed: Make sure your home phone is broadcasting with key: ${key}`;
      });
    }
  </script>
